# Requirements for the "ChangeTIP-newbb" environment.
#
# Use this when you want to run the DINOv3 and/or SAM2 backbones. The original
# "Change3D" env (torch 2.2.0+cu118) stays untouched and remains the home of
# Change3D-baseline + DINOv2 experiments.
#
# Install order (do NOT just pip install -r this file blindly — torch must
# come from the cu118 index, the rest from PyPI):
#
#   conda create -n ChangeTIP-newbb python=3.11 -y
#   conda activate ChangeTIP-newbb
#
#   # 1. torch from the cu118 index (matches your CUDA 11.8 driver)
#   pip install torch==2.5.1+cu118 torchvision==0.20.1+cu118 \
#       --index-url https://download.pytorch.org/whl/cu118
#
#   # 2. everything else from PyPI
#   pip install -r requirements-newbb.txt
#
#   # 3. SAM2 (last, --no-deps to keep our pinned torch)
#   pip install --no-deps git+https://github.com/facebookresearch/sam2.git
#
# Then verify:
#   python -c "
#   import torch, transformers
#   from transformers.utils import is_torch_available
#   print(torch.__version__, transformers.__version__, is_torch_available())
#   "
# Expected: 2.5.1+cu118  4.56.x (or higher)  True
